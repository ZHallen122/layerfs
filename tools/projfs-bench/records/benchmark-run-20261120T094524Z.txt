## 4. quiet-run.ps1

# Waits until no other build or test runs on the machine, then runs the whole
# benchmark from the clean clone on the repository's toolchain, and records
# every process of another session's build or test seen while it runs.
param(
    [Parameter(Mandatory)][string]$Clone,
    [Parameter(Mandatory)][string]$Log,
    [int]$QuietSamples = 15,
    [int]$GateSeconds = 20,
    [int]$WatchSeconds = 30,
    [int]$GateLimitMinutes = 360
)
$ErrorActionPreference = 'Stop'
$env:RUSTUP_TOOLCHAIN = '1.85.1'
function Say([string]$Line) { "$((Get-Date).ToUniversalTime().ToString('yyyy-MM-ddTHH:mm:ssZ')) $Line" | Add-Content -Encoding utf8 $Log }
function Suspects {
    # Builds and tests: cargo, rustc, and anything started from the main
    # checkout's target directory, where the other session's worktree lives.
    @(Get-CimInstance Win32_Process | Where-Object {
            $_.Name -in 'cargo.exe', 'rustc.exe' -or
            ($_.ExecutablePath -and $_.ExecutablePath -like '*\Open_source\layerfs\target\*')
        })
}
function Below([int]$Root, $All, $Process) {
    $at = $Process
    for ($depth = 0; $depth -lt 24 -and $at; $depth++) {
        if ($at.ProcessId -eq $Root) { return $true }
        $at = $All[[int]$at.ParentProcessId]
    }
    $false
}
Say "gate: waiting for $QuietSamples samples in a row, $GateSeconds s apart, with no build or test running"
$quiet = 0
$began = Get-Date
while ($quiet -lt $QuietSamples) {
    if (((Get-Date) - $began).TotalMinutes -gt $GateLimitMinutes) { Say 'gate: never quiet; nothing was run'; exit 3 }
    $seen = Suspects
    if ($seen.Count) {
        if ($quiet -ge 0) { Say "gate: busy ($(($seen | ForEach-Object Name | Sort-Object -Unique) -join ' '))" }
        $quiet = -1
    }
    $quiet = [Math]::Max(0, $quiet) + [int](-not $seen.Count)
    Start-Sleep -Seconds $GateSeconds
}
Say 'gate: quiet'
Set-Location $Clone
$work = Join-Path $env:TEMP 'lfsb'
if (Test-Path $work) { Rename-Item $work "lfsb-set-aside-$((Get-Date).ToString('HHmmss'))" }
Say "clean checkout at $(& git rev-parse HEAD), dirty: [$(& git status --porcelain)]"
Say "toolchain: $(& cargo --version) / $(& rustc --version)"
$run = Start-Process powershell -ArgumentList '-NoProfile', '-ExecutionPolicy', 'Bypass', '-File', 'engine\crates\layerfs-projfs\bench\run.ps1' `
    -NoNewWindow -PassThru -RedirectStandardOutput "$Log.run.out" -RedirectStandardError "$Log.run.err"
Say "benchmark started, process $($run.Id)"
$foreign = 0
while (-not $run.HasExited) {
    Start-Sleep -Seconds $WatchSeconds
    $all = @{}
    Get-CimInstance Win32_Process | ForEach-Object { $all[[int]$_.ProcessId] = $_ }
    foreach ($process in (Suspects)) {
        if (-not (Below $run.Id $all $process)) {
            $foreign++
            $text = "$($process.CommandLine)"
            Say "FOREIGN $($process.Name) $($process.ProcessId): $($text.Substring(0, [Math]::Min(160, $text.Length)))"
        }
    }
}
Say "benchmark exit: $($run.ExitCode); foreign build or test processes seen in samples: $foreign"
Say "dirty after the benchmark: [$(& git status --porcelain)]"
Say 'finished'
